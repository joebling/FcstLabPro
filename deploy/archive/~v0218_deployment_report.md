## 六、策略代码

```python
# v0218 策略逻辑
import pandas as pd
import numpy as np

# 加载 Bull 模型预测
bull_pred = pd.read_csv('experiments/weekly/weekly_bull_v27_orion_0218/predictions.csv')
prices_df = pd.read_csv('data/raw/btc_binance_BTCUSDT_1d.csv')

# 计算三重MA
prices_df['sma_50'] = prices_df['close'].rolling(50).mean()
prices_df['sma_150'] = prices_df['close'].rolling(150).mean()
prices_df['sma_200'] = prices_df['close'].rolling(200).mean()
prices_df['above_ma50'] = (prices_df['close'] > prices_df['sma_50']).astype(int)
prices_df['above_ma150'] = (prices_df['close'] > prices_df['sma_150']).astype(int)
prices_df['above_ma200'] = (prices_df['close'] > prices_df['sma_200']).astype(int)

n = len(bull_pred)
y_bull = bull_pred['y_pred'].values
hold_days = 13

# 生成交易信号
y_signal = np.zeros(n)
current_pos = 0
days_in_pos = 0

for i in range(n):
    if days_in_pos > 0:
        # 持有仓位
        y_signal[i] = current_pos
        days_in_pos -= 1
    elif y_bull[i] == 1:
        # 三重MA过滤
        if (prices_df['above_ma50'].iloc[i] == 1 and
            prices_df['above_ma150'].iloc[i] == 1 and
            prices_df['above_ma200'].iloc[i] == 1):
            y_signal[i] = 1
            current_pos = 1
            days_in_pos = hold_days
    else:
        y_signal[i] = 0
        current_pos = 0
```
